# =====================================================================================
# allrepo_lite_selftest.py  -  BUILD 2026-11-16f
# Selftest for allrepo_lite_run.py. Builds a small synthetic full gold in a temp folder (never touches G: or H:),
# runs the bake on it and asserts on the output AND on what each run touches (file mtime_ns, files built / removed).
# The core / extension split is checked against an independent restatement of the rule written here (not the bake's
# own mask), and core + extension must give back the gold rows exactly (every column, every row).
# Run in the code folder, pinned environment:  python allrepo_lite_selftest.py      -> "SELFTEST n/n PASS", exit 0
# =====================================================================================

def check(name, cond, detail=""):
    RESULTS.append((name, bool(cond), detail))


def reset_config():
    C.DROP_COLUMNS[:] = ORIG["DROP"]
    C.ROUND_COLUMNS.clear()
    C.ROUND_COLUMNS.update(ORIG["ROUND"])
    C.ROUND_STORE = ORIG["STORE"]
    C.ROW_GROUP_ROWS = ORIG["RG"]
    C.WINDOW_SLA[:] = ORIG["WSLA"]
    C.THIN_CATEGORIES[:] = ORIG["THIN"]
    C.RECENT_DAYS = ORIG["DAYS"]
    R.SLICE_ROWS = ORIG["SLICE"]
    R.REPLACE_WAIT_S = ORIG["WAIT"]


def small():
    C.ROW_GROUP_ROWS = 256            # several row groups per lite file
    R.SLICE_ROWS = 100                # slices that cross gold row groups


# ----------------------------------------------------------------------------------- synthetic gold
SPECIAL = [1499.999, 1500.0, 2500.0, -400.0, -500.0, 499.9, None, 123456789.4, -2500.0, 0.0]


def gold_table(year, month, n, with_source, seed=0, bump=0.0, days=20):
    dates = [dt.date(year, month, 1 + (i % days)) for i in range(n)]
    vals = [SPECIAL[i % len(SPECIAL)] if i < 40 else ((i * 7919 + seed) % 10_000_000) / 7.0 - 300_000 for i in range(n)]
    vals = [v + bump if v is not None else None for v in vals]
    cols = {
        "SnapshotDate": pa.array(dates, pa.date32()),
        "HeadOfficeSirisId": pa.array(["%010d" % (i % 13) for i in range(n)]),
        "SLA_CateCode": pa.array([SLAS[(i // 3) % len(SLAS)] for i in range(n)]),
        "Category": pa.array([CATS[(i // 7) % len(CATS)] for i in range(n)]),
        "AmountBuy": pa.array([float(i) for i in range(n)]),
        "AmountSell": pa.array([float(-i) for i in range(n)]),
        "IA_EUR": pa.array([i * 1.123456789 for i in range(n)]),
        "Haircut": pa.array([float("nan") if i % 17 == 0 else (None if i % 19 == 0 else (i % 100) / 1000.0)
                             for i in range(n)], pa.float64()),
        "MaturityDate": pa.array([dt.date(2073, 1, 1) if i % 11 == 0 else (None if i % 23 == 0 else dt.date(2027, 5, 3))
                                  for i in range(n)], pa.date32()),
        "CCCTradeKey": pa.array([None if i % 29 == 0 else ("" if i % 31 == 0 else "k%d" % i) for i in range(n)]),
        "IssuerId": pa.array(["000000000#" if i % 5 == 0 else "%010d" % i for i in range(n)]),
        "Trades": pa.array(list(range(n)), pa.int64()),          # row number: puts core + ext back in gold order
        "FO_ID": pa.array(["fo%d" % i for i in range(n)]),
        "MarkToMark_EUR": pa.array([float(i) * 3.3 for i in range(n)]),
        "MTMCash": pa.array([float(i) * 1.7 for i in range(n)]),
        "MTMSecurity": pa.array([float(i) * 1.9 for i in range(n)]),
        "SumAmountBuySell_EUR": pa.array([float(i) * 2.1 for i in range(n)]),
        "TransactionId": pa.array(["t%d" % i for i in range(n)]),
    }
    for j, c in enumerate(ROUND):
        cols[c] = pa.array([None if v is None else v * (1 + j) for v in vals], pa.float64())
    if with_source:
        cols["_SourceFile"] = pa.array(["Extraction_AllRepo_PROD_Q%04d%02d%02d.csv" % (year, month, 1 + (i % days))
                                        for i in range(n)])
    return pa.table(cols)


def write_gold(path, t, rg=None):
    pq.write_table(t, path, row_group_size=rg or max(t.num_rows, 1), compression="snappy")


def make_gold(root):
    h = os.path.join(root, "gold", "ALLREPO_gold")
    c = os.path.join(root, "gold", "ALLREPO_gold_v2", "current")
    os.makedirs(h)
    os.makedirs(c)
    write_gold(os.path.join(h, "FACT_ALLREPO_gold_2024.parquet"), gold_table(2024, 3, 700, False, 1), rg=300)
    write_gold(os.path.join(h, "FACT_ALLREPO_gold_2025.parquet"), gold_table(2025, 6, 650, False, 2))
    write_gold(os.path.join(c, "FACT_ALLREPO_gold_2026-09.parquet"), gold_table(2026, 9, 400, True, 3))
    write_gold(os.path.join(c, "FACT_ALLREPO_gold_2026-10.parquet"), gold_table(2026, 10, 450, True, 4))
    with open(os.path.join(h, "notes.txt"), "w") as fh:
        fh.write("not a gold file")
    ids = os.path.join(root, "ids.csv")
    with open(ids, "w", encoding="utf-8") as fh:
        fh.write('HeadOfficeSirisId\n0000000005\n7\n"0000000009"\n\n')
    return h, c, ids


def run(h, c, lite, ids, *extra):
    buf = io.StringIO()
    args = ["--hist-dir", h, "--current-dir", c, "--lite-root", lite] + (["--ids-file", ids] if ids else [])
    with contextlib.redirect_stdout(buf):
        code = R.main(args + list(extra))
    return code, buf.getvalue()


def lite_files(lite):
    out = {}
    for side in R.DATA_FOLDERS:
        d = os.path.join(lite, side)
        if os.path.isdir(d):
            for n in os.listdir(d):
                out[os.path.join(side, n)] = os.stat(os.path.join(d, n)).st_mtime_ns
    return out


def built_lines(text):
    return [l for l in text.splitlines() if l.startswith("Built: ")]


def expected_round(v, nd=-3):
    if v is None:
        return None
    return float(round(v / 10.0 ** (-nd)) * 10.0 ** (-nd)) + 0.0      # Python round = half to even


def all_gold_dates(h, c):
    ds = set()
    for d in (h, c):
        for n in os.listdir(d):
            if n.endswith(".parquet"):
                ds.update(pq.read_table(os.path.join(d, n), columns=["SnapshotDate"]).column(0).to_pylist())
    return ds


def rule_core(row, window, ids):
    """Independent restatement of LITE CONFIG v1.3 (not the bake's code)."""
    sla, cat, cid, d = row["SLA_CateCode"], row["Category"], row["HeadOfficeSirisId"], row["SnapshotDate"]
    win = d in window
    if sla == "07L":
        thinned = cat == "FCG21" or cid in ids
        return (not thinned) or win
    if sla == "07K":
        return win
    return False


def window_of(dates, days=90):
    anchor = max(dates)
    last = {}
    for d in dates:
        last[(d.year, d.month)] = max(d, last.get((d.year, d.month), d))
    return set(last.values()) | {d for d in dates if (anchor - d).days <= days}


def check_pair(tag, gold_path, lite, rel, window, ids=IDS_IN_FILE):
    """core + ext = gold rows; each row in the part the rule gives it; values as the lite rules say."""
    g = pq.read_table(gold_path)
    core = pq.read_table(os.path.join(lite, rel))
    ext = pq.read_table(os.path.join(lite, "ext_" + rel))
    rows = g.to_pylist()
    want_core = [r["Trades"] for r in rows if rule_core(r, window, ids)]
    want_ext = [r["Trades"] for r in rows if not rule_core(r, window, ids)]
    check(tag + " core + ext rows = gold rows", core.num_rows + ext.num_rows == g.num_rows,
          (core.num_rows, ext.num_rows, g.num_rows))
    check(tag + " rows in the part the rule gives them (core %d, ext %d)" % (len(want_core), len(want_ext)),
          core.column("Trades").to_pylist() == want_core and ext.column("Trades").to_pylist() == want_ext)
    both = pa.concat_tables([core, ext]).sort_by("Trades")
    ok = True
    for x in g.column_names:
        if x in DROPPED:
            ok = ok and x not in both.column_names
        elif x in C.ROUND_COLUMNS:
            ok = ok and both.column(x).to_pylist() == [expected_round(v) for v in g.column(x).to_pylist()]
        else:
            ok = ok and R.same_values(pa, pc, both.column(x), g.column(x))
    check(tag + " core + ext = gold values (copied identical, rounded = rounding, drops gone)", ok)
    return core, ext


# ----------------------------------------------------------------------------------- tests
def tests(root):
    h, c, ids = make_gold(root)
    lite = os.path.join(root, "out", "ALLREPO_lite")
    small()
    win = window_of(all_gold_dates(h, c))

    # T1 first run
    code, out = run(h, c, lite, ids)
    f1 = lite_files(lite)
    names = ["FACT_ALLREPO_lite_2024.parquet", "FACT_ALLREPO_lite_2025.parquet"]
    cur = ["FACT_ALLREPO_lite_2026-09.parquet", "FACT_ALLREPO_lite_2026-10.parquet"]
    expect = [os.path.join(s, n) for s in ("hist", "ext_hist") for n in names] + \
             [os.path.join(s, n) for s in ("current", "ext_current") for n in cur]
    check("T1 first run exit 0", code == 0, out[-500:])
    check("T1 eight lite files: core + extension per gold file", sorted(f1) == sorted(expect), sorted(f1))
    check("T1 four pairs built", len(built_lines(out)) == 4)
    check("T1 intra-group ids: 3 listed, 1 padded, found 3", "listed 3 (1 padded" in out and "found in the gold 3" in out,
          [l for l in out.splitlines() if l.startswith("intra-group")])

    # T2-T4 every pair against the gold and the independent rule
    for rel, gp in ((os.path.join("hist", names[0]), os.path.join(h, "FACT_ALLREPO_gold_2024.parquet")),
                    (os.path.join("hist", names[1]), os.path.join(h, "FACT_ALLREPO_gold_2025.parquet")),
                    (os.path.join("current", cur[0]), os.path.join(c, "FACT_ALLREPO_gold_2026-09.parquet")),
                    (os.path.join("current", cur[1]), os.path.join(c, "FACT_ALLREPO_gold_2026-10.parquet"))):
        core, ext = check_pair("T2 " + os.path.basename(rel), gp, lite, rel, win)
    core = pq.read_table(os.path.join(lite, "hist", names[0]))
    ext = pq.read_table(os.path.join(lite, "ext_hist", names[0]))
    g = pq.read_table(os.path.join(h, "FACT_ALLREPO_gold_2024.parquet"))
    check("T2 hist: CCP 07L rows in core only on the month-end", all(
        r["SnapshotDate"] == dt.date(2024, 3, 20) for r in core.to_pylist()
        if r["Category"] == "FCG21" and r["SLA_CateCode"] == "07L"))
    check("T2 hist: intra-group 07L rows in core only on the month-end", all(
        r["SnapshotDate"] == dt.date(2024, 3, 20) for r in core.to_pylist()
        if r["HeadOfficeSirisId"] in IDS_IN_FILE and r["SLA_CateCode"] == "07L"))
    check("T2 hist: other 07L rows in core on every date", sorted({r["SnapshotDate"] for r in core.to_pylist()
          if r["SLA_CateCode"] == "07L" and r["Category"] != "FCG21" and r["HeadOfficeSirisId"] not in IDS_IN_FILE})
          == sorted(set(g.column("SnapshotDate").to_pylist())))
    check("T2 hist: 07K in core only on the month-end", {r["SnapshotDate"] for r in core.to_pylist()
          if r["SLA_CateCode"] == "07K"} == {dt.date(2024, 3, 20)})
    check("T2 07F / 07Y / null SLA never in core", not [r for r in core.to_pylist()
          if r["SLA_CateCode"] not in ("07L", "07K")])
    check("T2 null Category + 07L = not thinned (core every date)", any(r["Category"] is None and r["SLA_CateCode"] == "07L"
          and r["SnapshotDate"] != dt.date(2024, 3, 20) and r["HeadOfficeSirisId"] not in IDS_IN_FILE
          for r in core.to_pylist()))
    cc = pq.read_table(os.path.join(lite, "current", cur[0])).to_pylist()
    gc = pq.read_table(os.path.join(c, "FACT_ALLREPO_gold_2026-09.parquet")).to_pylist()
    ccp = lambda rows: sorted(r["Trades"] for r in rows if r["Category"] == "FCG21" and r["SLA_CateCode"] == "07L")
    k07 = lambda rows: sorted(r["Trades"] for r in rows if r["SLA_CateCode"] == "07K")
    check("T2 current month inside the 90 days: every CCP 07L and 07K row kept in core",
          ccp(cc) == ccp(gc) and k07(cc) == k07(gc) and len({r["SnapshotDate"] for r in gc if r["SLA_CateCode"] == "07K"}) > 1)
    check("T2 core and ext share one schema", core.schema.equals(ext.schema))
    check("T2 several row groups", pq.ParquetFile(os.path.join(lite, "hist", names[0])).metadata.num_row_groups >= 1
          and pq.ParquetFile(os.path.join(lite, "ext_hist", names[0])).metadata.num_row_groups >= 2)
    both = pa.concat_tables([core, ext]).sort_by("Trades")
    got = both.column("RiskNet_EUR").to_pylist()
    check("T3 1500 -> 2000, 2500 -> 2000, -2500 -> -2000, 1499.999 -> 1000, null kept",
          got[1] == 2000.0 and got[2] == 2000.0 and got[8] == -2000.0 and got[0] == 1000.0 and got[6] is None, got[:10])
    check("T3 no -0.0", not [v for v in got if v == 0.0 and str(v).startswith("-")] and got[3] == 0.0 and got[4] == 0.0)
    check("T3 rounded columns double", all(core.schema.field(x).type == pa.float64() for x in ROUND))
    check("T4 hist _SourceFile = gold yearly file name (core and ext)",
          set(both.column("_SourceFile").to_pylist()) == {"FACT_ALLREPO_gold_2024.parquet"})
    lc = pq.read_table(os.path.join(lite, "current", cur[1]))
    check("T4 hist and current lite files have the same column set and types",
          sorted(lc.column_names) == sorted(core.column_names)
          and all(lc.schema.field(x).type == core.schema.field(x).type for x in core.column_names))
    tot = [l for l in out.splitlines() if l.startswith("Rows in the lite folders")]
    check("T4 report gives core + extension totals", tot and "core" in tot[0], tot)

    # T5 second run touches nothing
    time.sleep(0.02)
    code, out = run(h, c, lite, ids)
    check("T5 second run exit 0, nothing built", code == 0 and not built_lines(out), out[-300:])
    check("T5 no lite file touched (mtime_ns)", lite_files(lite) == f1)

    # T6 one gold month changes -> only that pair rebuilt
    write_gold(os.path.join(c, "FACT_ALLREPO_gold_2026-10.parquet"), gold_table(2026, 10, 470, True, 4, bump=1.0))
    code, out = run(h, c, lite, ids)
    f2 = lite_files(lite)
    changed = sorted(k for k in f2 if f2[k] != f1.get(k))
    check("T6 only the 2026-10 pair rebuilt", changed == [os.path.join("current", cur[1]),
                                                          os.path.join("ext_current", cur[1])], changed)
    check("T6 reason gold changed", any("(gold changed)" in x for x in built_lines(out)))
    check_pair("T6 2026-10", os.path.join(c, "FACT_ALLREPO_gold_2026-10.parquet"), lite,
               os.path.join("current", cur[1]), win)

    # T6b gold file rewritten with the same size (same data, new mtime) -> rebuilt
    p10 = os.path.join(c, "FACT_ALLREPO_gold_2026-10.parquet")
    size10 = os.path.getsize(p10)
    write_gold(p10, pq.read_table(p10))
    os.utime(p10, ns=(time.time_ns(), time.time_ns() + 5_000_000_000))
    code, out = run(h, c, lite, ids)
    check("T6b same-size gold rewrite rebuilt (mtime_ns key)", os.path.getsize(p10) == size10
          and len(built_lines(out)) == 1 and "2026-10" in built_lines(out)[0])

    # T13 lite files edited / removed outside the job -> rebuilt (core and extension)
    p = os.path.join(lite, "current", cur[0])
    pq.write_table(pq.read_table(p).slice(0, 10), p)
    code, out = run(h, c, lite, ids)
    check("T13 core file changed outside -> rebuilt", any("lite file changed outside" in x for x in built_lines(out)))
    pe = os.path.join(lite, "ext_current", cur[0])
    pq.write_table(pq.read_table(pe).slice(0, 3), pe)
    code, out = run(h, c, lite, ids)
    check("T13 extension file changed outside -> rebuilt", any("lite file changed outside" in x for x in built_lines(out)))
    os.remove(pe)
    code, out = run(h, c, lite, ids)
    check("T13 extension file missing -> rebuilt", any("lite file missing" in x for x in built_lines(out))
          and os.path.exists(pe))
    check_pair("T13 2026-09", os.path.join(c, "FACT_ALLREPO_gold_2026-09.parquet"), lite,
               os.path.join("current", cur[0]), win)

    # T7 config change -> every pair rebuilt
    f3 = lite_files(lite)
    del C.ROUND_COLUMNS["MTMCash_EUR"]
    code, out = run(h, c, lite, ids)
    f4 = lite_files(lite)
    check("T7 config change rebuilds all 4 pairs", code == 0 and len(built_lines(out)) == 4
          and all(f4[k] != f3[k] for k in f3), out[-300:])
    check("T7 reason config changed", all("(config changed)" in x for x in built_lines(out)))
    l7 = pa.concat_tables([pq.read_table(os.path.join(lite, "hist", names[0])),
                           pq.read_table(os.path.join(lite, "ext_hist", names[0]))]).sort_by("Trades")
    check("T7 MTMCash_EUR back to full precision", l7.column("MTMCash_EUR").equals(g.column("MTMCash_EUR")))
    reset_config()
    small()
    code, out = run(h, c, lite, ids)
    check("T7 back to the config rebuilds all 4", len(built_lines(out)) == 4)

    # T24 split change: 07K no longer kept in core -> all rebuilt, 07K only in the extension
    C.WINDOW_SLA[:] = []
    code, out = run(h, c, lite, ids)
    c24 = pq.read_table(os.path.join(lite, "current", cur[0]))
    check("T24 split config change rebuilds all and moves 07K out of core", len(built_lines(out)) == 4
          and not [r for r in c24.to_pylist() if r["SLA_CateCode"] == "07K"])
    reset_config()
    small()
    run(h, c, lite, ids)

    # T25 intra-group id list change -> all rebuilt; ids file problems FAIL before writing
    before = lite_files(lite)
    ids2 = os.path.join(root, "ids2.csv")
    with open(ids2, "w") as fh:
        fh.write("0000000005\n")
    code, out = run(h, c, lite, ids2)
    check("T25 id list change rebuilds all pairs", code == 0 and len(built_lines(out)) == 4
          and all("(config changed)" in x for x in built_lines(out)))
    check_pair("T25 2026-09 with one id", os.path.join(c, "FACT_ALLREPO_gold_2026-09.parquet"), lite,
               os.path.join("current", cur[0]), win, ids=["0000000005"])
    run(h, c, lite, ids)
    before = lite_files(lite)
    bad = os.path.join(root, "ids_bad.csv")
    with open(bad, "w") as fh:
        fh.write("0000000005\nCPY-12\n")
    code, out = run(h, c, lite, bad)
    check("T25 non-digit id FAILS, nothing written", code == 2 and lite_files(lite) == before and "CPY-12" in out)
    code, out = run(h, c, lite, os.path.join(root, "no_such.csv"))
    check("T25 missing id file FAILS, nothing written", code == 2 and lite_files(lite) == before)

    # T8 int64 store (L6 test) into a scratch root
    scratch = os.path.join(root, "out", "L6_int64")
    code, out = run(h, c, scratch, ids, "--only", "2026-10", "--round-store", "int64")
    li = pa.concat_tables([pq.read_table(os.path.join(scratch, "current", cur[1])),
                           pq.read_table(os.path.join(scratch, "ext_current", cur[1]))]).sort_by("Trades")
    ld = pa.concat_tables([pq.read_table(os.path.join(lite, "current", cur[1])),
                           pq.read_table(os.path.join(lite, "ext_current", cur[1]))]).sort_by("Trades")
    check("T8 int64 run exit 0, one pair", code == 0 and sorted(lite_files(scratch)) ==
          sorted([os.path.join("current", cur[1]), os.path.join("ext_current", cur[1])]), out[-300:])
    check("T8 rounded columns int64", all(li.schema.field(x).type == pa.int64() for x in ROUND))
    check("T8 int64 values = double values", all(
        [None if v is None else float(v) for v in li.column(x).to_pylist()] == ld.column(x).to_pylist() for x in ROUND))
    check("T8 fingerprint differs by store", R.config_fingerprint("int64", "x") != R.config_fingerprint("double", "x"))
    old = C.LITE_ROOT
    C.LITE_ROOT = lite
    before = lite_files(lite)
    code, out = run(h, c, lite, ids, "--round-store", "int64")
    C.LITE_ROOT = old
    check("T8 --round-store on the production lite root refused", code == 2 and lite_files(lite) == before)

    # T16 --only
    before = lite_files(lite)
    code, out = run(h, c, lite, ids, "--only", "2025", "--full")
    after = lite_files(lite)
    check("T16 --only 2025 --full rebuilds that year's pair only",
          sorted(k for k in after if after[k] != before[k]) == sorted([os.path.join("hist", names[1]),
                                                                      os.path.join("ext_hist", names[1])]))
    code, out = run(h, c, lite, ids, "--only", "2019")
    check("T16 --only with an unknown period FAILS", code == 2)

    # T10 typo in config -> FAIL, nothing written
    before = lite_files(lite)
    C.DROP_COLUMNS.append("AmountBuyy")
    code, out = run(h, c, lite, ids, "--full")
    reset_config()
    small()
    check("T10 unknown config column FAILS before writing", code == 2 and lite_files(lite) == before
          and "AmountBuyy" in out)
    C.DROP_COLUMNS.append("SLA_CateCode")
    code, out = run(h, c, lite, ids, "--full")

    real_replace = os.replace

    def flaky(src, dst):

    # T15 verify catches a wrong value / a row in the wrong part -> FAIL, previous files kept, no tmp

    real_mask = R.core_mask
    state = {"n": 0}

    def drifting(pa_, pc_, t, w, i):                  # build sees one rule, verify another
        state["n"] += 1
        m = real_mask(pa_, pc_, t, w, i)
        return pc.invert(m) if state["n"] == 1 else m
    R.core_mask = drifting

        code, out = run(h, c, lite, ids, "--only", "2026-09", "--full")
    finally:

    write_gold(p25, keep25)

    check("T19 fixed gold file rebuilt next run", code == 0 and any("2025" in x for x in built_lines(out)))

def main():
    reset_config()
    root = tempfile.mkdtemp(prefix="allrepo_lite_selftest_")
    try:
        tests(root)
    except Exception as ex:
        import traceback
        traceback.print_exc()
        RESULTS.append(("crash: %s" % ex, False, ""))
    finally:
        reset_config()
        shutil.rmtree(root, ignore_errors=True)
    for name, ok, detail in RESULTS:
        print("%s  %s%s" % ("ok  " if ok else "FAIL", name, "" if ok else "  " + str(detail)[:300]))
    n_ok = sum(1 for _, ok, _ in RESULTS if ok)
    print("SELFTEST %d/%d %s  (allrepo_lite_run BUILD %s, selftest BUILD %s)"
          % (n_ok, len(RESULTS), "PASS" if n_ok == len(RESULTS) else "FAIL", R.BUILD, BUILD))
    return 0 if n_ok == len(RESULTS) else 1


if __name__ == "__main__":
    sys.exit(main())
